# Natural Earth data compiled into the library as byte arrays, so the offline
# basemap and place search need no data files next to the binary.
function(wetwaresuite_embed name file out)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
  file(READ ${file} _hex HEX)
  string(LENGTH "${_hex}" _hexlen)
  math(EXPR _bytes "${_hexlen} / 2")
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _arr "${_hex}")
  string(REGEX REPLACE "((0x..,){32})" "\\1\n" _arr "${_arr}")
  get_filename_component(_fname ${file} NAME)
  set(_src "// Generated from third_party/naturalearth/${_fname} (public domain). Do not edit.\n#include <cstddef>\nextern const unsigned char ${name}[];\nextern const std::size_t ${name}Size;\nconst unsigned char ${name}[] = {\n${_arr}0};\nconst std::size_t ${name}Size = ${_bytes};\n")
  if(EXISTS ${out})
    file(READ ${out} _old)
  else()
    set(_old "")
  endif()
  if(NOT _old STREQUAL _src)
    file(WRITE ${out} "${_src}")
  endif()
endfunction()
set(_countries_cpp ${CMAKE_CURRENT_BINARY_DIR}/countries_data.cpp)
set(_places_cpp ${CMAKE_CURRENT_BINARY_DIR}/places_data.cpp)
wetwaresuite_embed(kWetwaresuiteCountries ${PROJECT_SOURCE_DIR}/third_party/naturalearth/countries.geojson ${_countries_cpp})
wetwaresuite_embed(kWetwaresuitePlaces ${PROJECT_SOURCE_DIR}/third_party/naturalearth/places.geojson ${_places_cpp})

            src/track_commands.cpp src/nav_commands.cpp src/place_commands.cpp
            ${_countries_cpp} ${_places_cpp})
add_library(wetwaresuite::maps ALIAS wetwaresuite_maps)

target_link_libraries(wetwaresuite_maps PUBLIC wetwaresuite::core wetwaresuite::raster PRIVATE wetwaresuite::pdf wetwaresuite::pdfdoc nlohmann_json::nlohmann_json)
