# The Natural Earth 110m countries are compiled into the library as a byte
# array, so the offline basemap needs no data files next to the binary.
set(_countries ${PROJECT_SOURCE_DIR}/third_party/naturalearth/countries.geojson)
set(_countries_cpp ${CMAKE_CURRENT_BINARY_DIR}/countries_data.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_countries})
file(READ ${_countries} _hex HEX)
string(LENGTH "${_hex}" _hexlen)
math(EXPR _bytes "${_hexlen} / 2")
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _arr "${_hex}")
string(REGEX REPLACE "((0x..,){32})" "\\1\n" _arr "${_arr}")
set(_src "// Generated from third_party/naturalearth/countries.geojson (public domain). Do not edit.\n#include <cstddef>\nextern const unsigned char kWetwaresuiteCountries[];\nextern const std::size_t kWetwaresuiteCountriesSize;\nconst unsigned char kWetwaresuiteCountries[] = {\n${_arr}0};\nconst std::size_t kWetwaresuiteCountriesSize = ${_bytes};\n")
if(EXISTS ${_countries_cpp})
  file(READ ${_countries_cpp} _old)
else()
  set(_old "")
endif()
if(NOT _old STREQUAL _src)
  file(WRITE ${_countries_cpp} "${_src}")
endif()

add_library(wetwaresuite_maps src/geo.cpp src/analysis.cpp src/io.cpp src/shapefile.cpp src/raster_io.cpp src/render.cpp src/commands.cpp
            ${_countries_cpp})
add_library(wetwaresuite::maps ALIAS wetwaresuite_maps)
target_include_directories(wetwaresuite_maps PUBLIC include)
target_link_libraries(wetwaresuite_maps PUBLIC wetwaresuite::core wetwaresuite::raster PRIVATE wetwaresuite::pdf wetwaresuite::pdfdoc nlohmann_json::nlohmann_json)

if(WETWARESUITE_BUILD_TESTS)
  add_executable(test_maps tests/test_maps.cpp)
  target_link_libraries(test_maps PRIVATE wetwaresuite::maps wetwaresuite::mcp wetwaresuite_check)
  add_test(NAME maps COMMAND test_maps)
endif()
